        let buildPath = "/tmp/screenrec-21f3-final-native-build/arm64-apple-macosx/debug/FluidAudio_FluidAudio.bundle"

        let preferredBundle = Bundle(path: mainPath)

        guard let bundle = preferredBundle ?? Bundle(path: buildPath) else {

        return bundle
    }()
